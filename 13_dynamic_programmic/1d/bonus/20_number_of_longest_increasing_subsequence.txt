def longestIncreasingSubsequence(nums: list[int]) -> int:
	if not nums:
		return 0

	minVal = nums[0]
	maxCount = 0
	count = 1
	for i in range(1, len(nums)):
		num = nums[i]
		if num < minVal:
			minVal = num
            count = 1
            maxCount = max(maxCount, count)
        else:
        count += 1
        
    return max(maxCount, count)
