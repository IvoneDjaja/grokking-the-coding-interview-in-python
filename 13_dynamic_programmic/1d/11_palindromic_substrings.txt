def palindromicSubstrings(s: str) -> int:
   	n = len(s)
   	count = 0
    def expand(start, end):
        while start >= 0 and end < n and s[start] == s[end]:
            start -= 1
            end += 1
        length = end - start - 1
        return length
    for i in range(n):
        count += 1 if expand(i, i) > 0 else 0
        count += 1 if expand(i, i + 1) > 0 else 0
    return count
