def coinChange2(coins: List[int], amount: int) -> int:
    n = len(coins)
    dp = [[0] * (amount + 1) for _ in range(n + 1)]

    for i in range(1, n + 1):
        coin = coins[i]
        for j in range(1, amount + 1):
        if j % coin == 0 or dp[i][j-coin] > 0:
            dp[i][j] = dp[i-1][j] + 1 + dp[i][j-coin]
