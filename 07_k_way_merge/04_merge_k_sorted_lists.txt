def merge_k_lists(lists):
    if len(lists) == 0:
	  return None
    merged = [lists[0]]
    for i in range(1, len(lists)):
        list1 = merged.pop()
	   list2 = lists[i]
        merged.append(merge_two_lists(list1, list2))

    return merged.pop()
