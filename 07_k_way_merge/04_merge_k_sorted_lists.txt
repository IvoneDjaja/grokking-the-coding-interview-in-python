import heapq
def merge_k_sorted_lists(lists: ListNode) -> ListNode:
    min_heap = []
    heapq.heapify(min_heap)

    for list in lists:
        current = list
        while current is not None:
            heapq.heappush(min_heap, [current.val, current])
            current = current.next

    head = ListNode()
    current = head
    while min_heap:
        next = heapq.heappop(min_heap)
    current.next = next[1]
    current = next
        
    return head.next
